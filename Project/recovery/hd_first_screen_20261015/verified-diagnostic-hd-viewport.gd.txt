func _input(event:InputEvent) -> void:
	if event is InputEventMouseButton and event.pressed and is_instance_valid(game):
		if not Rect2(display.position,display.size).has_point(event.position):return
		var physical=(event.position-display.position)/display.size*Vector2(raster.size)
		var logical=raster.get_stretch_transform().affine_inverse()*physical
		assert(logical.is_equal_approx((event.position-display.position)/display.size*Vector2(1024,768)))
		var target=game.worlds[0].to_cell(logical)
		print("HD_REAL_MOUSE_MAPPING display=",event.position," raster=",physical," logical=",logical," cell=",target)
		if game.actor.can_enter(target):
			game.moving=game.actor.request(target)
			print("HD_REAL_CLICK_ROUTE_REQUEST ",game.moving," ",game.actor.path)
