func _ready() -> void:
	var reg=JSON.parse_string(FileAccess.get_file_as_string("res://assets/scifi/diagnostics/hero_hd4/manifest.json"))
	for group in reg.groups:
		var tex=load(group.atlas);assert(tex is Texture2D)
		for record in group.records:hd_atlases[int(record.source_id)]={"texture":tex,"region":record.region_px}
	assert(hd_atlases.size()==30)
	npc_hd=[load("res://assets/scifi/diagnostics/npc_hd4/fangzhang.png"),load("res://assets/scifi/diagnostics/npc_hd4/lingyunzi.png")]
	super._ready()
	add_child(hd_shadow_group)
	for i in 3:
		var sh=preload("res://scripts/art/density_alpha_ground_shadow.gd").new();hd_shadow_group.add_child(sh);hd_shadows.append(sh)
func draw_body() -> void:
	if not use_hd or not new_body:super.draw_body();return
	var data=hd_atlases[phase];var r=data.region
	var origin=worlds[0].screen_origin()+actor.position*Vector2(48,32)
	body.draw_texture_rect_region(data.texture,Rect2(origin-Vector2(16,64),Vector2(128,128)),Rect2(r[0],r[1],r[2],r[3]))
func _physics_process(delta:float) -> void:
	if moving:
		actor.step(delta)
		phase=actor.source_frame()
		if actor.path.is_empty():
			moving=false
			print("HD_REAL_CLICK_ROUTE_ARRIVED ",actor.position)
		refresh();label.visible=false

func install_hd_floor() -> void:
	var floor=preload("res://scripts/diagnostics/hd_floor.gd").new()
	assert(floor.setup(actor));assert(floor.prepare_hd())
	remove_child(worlds[0]);worlds[0].queue_free();worlds[0]=floor
	add_child(floor);move_child(floor,0);refresh()

func install_second_tree() -> void:
	var reg=worlds[0].read_json("res://assets/scifi/environment/a13_second_tree_reuse_v1/registration.json")
	assert(reg.owners.size()==7)
	for w in worlds+extra_passes:
		for owner in reg.owners:
			var order=int(owner.stable_source_order)
			assert(w.map.cells[order]==owner.source_cell)
			assert(not w.tree_owners.has(order))
			w._load_texture(owner.texture);assert(w.textures.has(owner.texture))
			w.tree_owners[order]=owner
	second_config=worlds[0].read_json("res://assets/scifi/environment/a13_second_tree_reuse_v1/shadow.json")
	second_shadow=Shadow.new();add_child(second_shadow);refresh()
func idle_base() -> int:
	return super.idle_base() if phase in PHASES else -100
func selected_texture() -> Texture2D:
	return super.selected_texture() if phase in PHASES else null
func refresh() -> void:
	new_body=prefer_new_body and phase in PHASES
	super.refresh()
	new_body=prefer_new_body and phase in PHASES
	body.texture_filter=CanvasItem.TEXTURE_FILTER_LINEAR if new_body else CanvasItem.TEXTURE_FILTER_NEAREST
	body.queue_redraw()
	if use_hd_npc and npc_on:
		for n in 2:
			var sprite=npc_pair.bodies[n];sprite.texture=npc_hd[n];sprite.scale=Vector2(.25,.25);sprite.region_rect=Rect2(0 if n==0 else ((npc_tick/17)%4)*512,0,512,640)
	if hd_shadows.size()==3:
		hd_shadow_group.visible=use_hd_shadows
		if use_hd_shadows:
			shadow.visible=false;npc_pair.visible=false
			var data=hd_atlases.get(phase,hd_atlases[13]);var r=data.region
			var origin=worlds[0].screen_origin()+actor.position*Vector2(48,32)-Vector2(16,64)
			assert(hd_shadows[0].configure_density_ground_frame(data.texture,Rect2i(int(r[0])/4,int(r[1])/4,128,128),4,Transform2D(0,origin),Vector2(40,86),0,Vector2(148.0/279,-111.0/279),.30,.9))
			hd_shadows[0].visible=new_body and shadow_on
			for n in 2:
				var cell=Vector2(64,37) if n==0 else Vector2(62,36)
				origin=worlds[0].screen_origin()+cell*Vector2(48,32)-Vector2(32,104)
				assert(hd_shadows[n+1].configure_density_ground_frame(npc_hd[n],Rect2i(0 if n==0 else ((npc_tick/17)%4)*128,0,128,160),4,Transform2D(0,origin),Vector2(44,118) if n==0 else Vector2(56,125),12.0/17 if n==0 else 0,Vector2(148.0/279,-111.0/279),.30,.9))
			move_child(hd_shadow_group,3)
	if is_instance_valid(second_shadow):
		var cfg=second_config;var reg=cfg.whole_tree_registration;var scale_value=float(reg.scale)
		var origin=Vector2(reg.world_top_left[0],reg.world_top_left[1])+worlds[0].screen_origin()
		assert(second_shadow.configure_ground_frame(load(cfg.texture),Rect2i(0,0,1254,1254),Transform2D(Vector2(scale_value,0),Vector2(0,scale_value),origin),Vector2(498,1224),0,Vector2(148.0/279,-111.0/279),.30,3))
		second_shadow.visible=tree_on and tree_cast_on;move_child(second_shadow,2)

	for item in vegetation_casts:
		item.node.position=worlds[0].screen_origin()+item.world_origin
		move_child(item.node,2)

func install_remaining_shrubs() -> void:
	var reg=worlds[0].read_json("res://art_source/a13_round_shrub_candidate_v1/technical_v1/first-screen-reuse-index.json")
	for group in reg.groups:
		if not Vector2i(int(group.center_owner_cell[0]),int(group.center_owner_cell[1])) in [Vector2i(64,45),Vector2i(51,44),Vector2i(58,51)]:continue
		for w in worlds+extra_passes:
			for owner in group.source_keys_and_sort:
				var order=int(owner.sort_tuple[2]);var frame=int(owner.key.split(":")[1].split("@")[0])
				assert(w.map.cells[order][2]==2 and w.map.cells[order][3]==frame)
				var path="res://assets/scifi/diagnostics/shrub_hd/owner-%d.png"%frame
				w._load_texture(path)
				w.tree_owners[order]={"texture":path,"world_top_left":group.candidate_raw_world_origin,"scale":group.scale}
		print("HD_SHRUB_GROUP_INSTALLED ",group.center_owner_cell)
	refresh()

		prefer_new_body=not prefer_new_body;refresh();label.visible=false
